from app.database.oracle import connection

def get_student(student_id):
    cursor = connection.cursor()

    cursor.execute("""
        SELECT
            ID,
            NAME,
            DEPARTMENT
        FROM STUDENT
        WHERE ID = :id
    """, {"id": student_id})

    row = cursor.fetchone()

    if row:
        return {
            "id": row[0],
            "name": row[1],
            "department": row[2]
        }

    return None
